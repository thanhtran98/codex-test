function updateFileInfo() {
const fileInput = document.getElementById('submission_files');
const filesInfo = document.getElementById('selected-files-info');
if (fileInput?.files?.length > 0) {
const maxTotalSize = 500 * 1024 * 1024; // 500MB
const newFiles = [];

for (let i = 0; i < fileInput.files.length; i++) {
const file = fileInput.files[i];
const exists = selectedFiles.some(existingFile =>
existingFile.name === file.name && existingFile.size === file.size
);
if (!exists) {
newFiles.push({
file: file,
id: Date.now() + Math.random() + i
});
}
}
fileInput.value = '';
// Calculate total size
let totalSizeBytes = 0;
selectedFiles.forEach(({ file }) => {
totalSizeBytes += file.size;
});
newFiles.forEach(({ file }) => {
totalSizeBytes += file.size;
});
// Check total size
if (totalSizeBytes > maxTotalSize) {
alert('⚠️ Tổng dung lượng file vượt quá giới hạn 500MB!');
updateFilesDisplay();
return;
}
selectedFiles = [...selectedFiles, ...newFiles];
updateFilesDisplay();
} else if (selectedFiles.length === 0) {
filesInfo.style.display = 'none';
}
}

function postSubmissionForm(formData) {
const controller = new AbortController();
const timeoutId = setTimeout(() => controller.abort(), 600000);
return fetch(window.location.href, {
method: 'POST',
body: formData,
signal: controller.signal,
headers: {
'X-Requested-With': 'XMLHttpRequest'
}
})
.then(response => {
if (!response.ok) {
throw new Error('HTTP ' + response.status + ': ' + response.statusText);
}
return response.text();
})
.then(data => {
clearTimeout(timeoutId);
let responseData;
try {
responseData = JSON.parse(data);
} catch (e) {
responseData = { success: false, error: 'Server returned invalid response: ' + data.substring(0, 200) };
}
if (responseData.success === true) {
clearInterval(uploadProgressInterval);
uploadProgressInterval = null;
const progressFill = document.getElementById('progressFill');
const progressStatus = document.getElementById('progressStatus');
if (progressFill) progressFill.style.width = '100%';
if (progressStatus) progressStatus.textContent = '✅ Nộp bài thành công!';
const step3 = document.getElementById('step3');
const step4 = document.getElementById('step4');
if (step3) step3.textContent = '✅ Lưu dữ liệu...';
if (step4) step4.textContent = '✅ Hoàn tất...';
setTimeout(() => window.location.reload(), 1500);
} else {
resetSubmitUiOnError(responseData.error || 'Có lỗi xảy ra khi nộp bài. Vui lòng thử lại.');
}
})
.catch(error => {
clearTimeout(timeoutId);
if (error.name === 'AbortError') {
resetSubmitUiOnError('Upload timeout! Vui lòng thử lại.');
} else {
resetSubmitUiOnError(error.message || 'Có lỗi xảy ra khi nộp bài.');
}
});
}

function updateExamEndCountdown() {
const countdownElement = document.getElementById('exam-end-countdown');
if (!countdownElement) return;
const startTime = parseInt(countdownElement.dataset.startTime);
const endTime = parseInt(countdownElement.dataset.endTime);
const deadline = parseInt(countdownElement.dataset.deadline);
// Use server time for consistency
const serverTime = 1791252137;
const now = Math.floor(Date.now() / 1000);
const timeDiff = now - serverTime;
const adjustedNow = serverTime + timeDiff;
const deadlineRemaining = deadline - adjustedNow;
// Check deadline first
if (deadlineRemaining <= 0) {
countdownElement.textContent = 'Đã hết hạn nộp bài';
countdownElement.style.color = '#dc3545';
} else if (adjustedNow < startTime) {
// Chưa đến thời gian bắt đầu thi
const timeToStart = startTime - adjustedNow;
// Nếu countdown về 0 hoặc âm thì reload trang
if (timeToStart <= 1) {
window.location.reload();
return;
}
const days = Math.floor(timeToStart / 86400);
const hours = Math.floor((timeToStart % 86400) / 3600);
const minutes = Math.floor((timeToStart % 3600) / 60);
const seconds = timeToStart % 60;
let timeString = '';
if (days > 0) {
timeString = `${days} ngày ${hours} giờ ${minutes} phút`;
} else if (hours > 0) {
timeString = `${hours} giờ ${minutes} phút`;
} else if (minutes > 0) {
timeString = `${minutes} phút ${seconds} giây`;
} else {
timeString = `${seconds} giây`;
}
countdownElement.textContent = 'Bắt đầu sau ' + timeString;
countdownElement.style.color = '#6c757d';
} else if (adjustedNow >= endTime) {
// Đã kết thúc thời gian thi
countdownElement.textContent = 'Đã kết thúc';
countdownElement.style.color = '#ffc107';
} else {
// Đang trong thời gian thi - hiển thị thời gian còn lại
const remaining = endTime - adjustedNow;
const days = Math.floor(remaining / 86400);
const hours = Math.floor((remaining % 86400) / 3600);
const minutes = Math.floor((remaining % 3600) / 60);
const seconds = remaining % 60;
let timeString = '';
if (days > 0) {
timeString = `${days} ngày ${hours} giờ ${minutes} phút`;
} else if (hours > 0) {
timeString = `${hours} giờ ${minutes} phút`;
} else if (minutes > 0) {
timeString = `${minutes} phút ${seconds} giây`;
} else {
timeString = `${seconds} giây`;
}
countdownElement.textContent = 'Còn lại ' + timeString;
// Change color based on remaining time
if (remaining < 300) { // Less than 5 minutes
countdownElement.style.color = '#dc3545';
} else if (remaining < 900) { // Less than 15 minutes
countdownElement.style.color = '#ffc107';
} else {
countdownElement.style.color = '#28a745';
}
}
}
